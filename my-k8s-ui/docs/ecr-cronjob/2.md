#!/bin/sh
set -e

REGISTRY="<AWSアカウントID>.dkr.ecr.ap-northeast-1.amazonaws.com"
CHART_NAME="my-chart"
VERSION="1.0.0"

# 1. CronJobが作成した Secret から ECR トークン（password）を取得してデコード
TOKEN=$(kubectl get secret ecr-helm-repo-secret -n argocd -o jsonpath='{.data.password}' | base64 -d)

# 2. Helm で ECR にログイン
echo "$TOKEN" | helm registry login \
  --username AWS \
  --password-stdin ${REGISTRY}

# 3. ECR上の Helm チャートを参照してインストール / アップグレード
# ※ ECR（OCI）を参照する際は先頭に oci:// を付与します
helm upgrade --install my-app oci://${REGISTRY}/${CHART_NAME} \
  --version ${VERSION} \
  --namespace default
